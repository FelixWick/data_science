\section{Combinatorics}

\subsection{Introduction}

\begin{frame}
\frametitle<presentation>{Introduction}
%
\begin{itemize}
	\item \alert{Counting Combinatorics:} A branch of combinatorics that deals with forming possible arrangements from $n$ elements.
	\item \alert{Combinatorial Models:} Determine the number of possible arrangements. 
	\item \alert{Important considerations:}
	\begin{itemize}
		\item Are all elements ($k=n$) or only some ($k<n$) of the elements used?
		\item Are the elements distinguishable?
		\item Is the order of elements relevant?
	\end{itemize}
	\item There is the classical \alert{urn model} in which all $n$ elements are placed in one urn and $k$ elements are drawn from it (a \alert{sample}), and a generalized \alert{balls-and-boxes} model, in which $k$ balls are distributed among $n$ boxes.
	\item Counting combinatorics forms the basis for calculating with \alert{classical probabilities} (see later).
\end{itemize}
%
\end{frame}

\section{Basics of Probability Theory}

\subsection{Events, Sample Space \& Event Set}

\begin{frame}[allowframebreaks,allowdisplaybreaks]
	\frametitle<presentation>{Events, Sample Space \& Event Set}
	%
	The starting point for probability theory is the concept of a \alert{random experiment}.
	It is also referred to as a \alert{random process} or a \alert{random observation}.
	
	\smallskip
	%
	\begin{alertblock}{Definition: Random Experiment}
		\begin{myframe}{
				An experiment is called a \textbf{random experiment} if it
				\begin{enumerate}
					\item is carried out according to a well-defined procedure,
					\item can be repeated arbitrarily often under the same conditions,
					\item has an outcome that is uncertain in advance and cannot be predicted.
				\end{enumerate}
			}
		\end{myframe}
	\end{alertblock}
	
	\begin{alertblock}{Examples}
	\begin{itemize}
		\item Classical random experiments: rolling a die, flipping a coin, drawing a card, spinning a roulette or wheel
		\item Even physical measurements can be seen as random experiments, e.g.
		\begin{itemize}
			\item current $I$ with voltage $U=230\,\text{V}$ and resistance $R=800\,\Omega$
			\item expected value from Ohm's law: 
			$I=U/R=0.2875\,\text{A}$
			\item exact outcome cannot be predicted precisely
			since it is influenced by environmental factors (e.g., temperature, humidity)
			\item measurement and reading errors introduce additional uncertainty
		\end{itemize}
	\end{itemize}
    \end{alertblock}
	%
	\begin{alertblock}{Definition: Elementary Event}
		\begin{myframe}{
				The individual outcomes of a random experiment that cannot be decomposed further and are mutually exclusive are called \textbf{elementary events}.
				%\marginpar{{\sffamily\mdseries\upshape Grundgesamtheit}}
			}
		\end{myframe}
	\end{alertblock}
	
	\alert{Example}
	In a \emph{die roll}, elementary outcomes are given by the number of pips on the top face.
	The experiment has six possible elementary events: $1, 2, 3, 4, 5, 6$.
	
	\smallskip
	
	\begin{alertblock}{Definition: Sample Space}
		\begin{myframe}{
				The set $\Omega$ of all elementary events of a random experiment is called the \textbf{event space}, \textbf{sample space}, or \textbf{basic set} of the random experiment.
				%\marginpar{{\sffamily\mdseries\upshape Grundgesamtheit}}
			}
		\end{myframe}
	\end{alertblock}
	
	\alert{Example}
	In the random experiment \emph{rolling a die} the sample space is $\Omega=\{1,2,3,4,5,6\}$. The sample space contains a finite number of elements.
	
	\smallskip
	%
	%\framebreak
	%
	\begin{alertblock}{Definition: Random Event}
		\begin{myframe}{
				A \textbf{random event} $A$ is a subset of the sample space $\Omega$. The (concrete) event $A$ has occurred if the outcome of the random experiment is an element of this subset $A$.
				%\marginpar{{\sffamily\mdseries\upshape Grundgesamtheit}}
			}
		\end{myframe}
	\end{alertblock}
	
	\alert{Example}
	In the experiment \emph{rolling two dice}, consider the random event $A = \{\text{sum of pips is greater than } 10\}$. This event consists of three elementary outcomes: 
	$A=\{(5,6),(6,5),(6,6)\}$.
	%
	\begin{alertblock}{Notation} 
		The individual elementary events are often written in set notation $\{\ast\}$ as follows:
		\begin{equation*}
			\label{eq:8-1}
			\Omega = \left\{ e_1, e_2, \ldots, e_m, \ldots \right\}
		\end{equation*}
		This notation can also be used for any (concrete) event $A$, e.g., $A = \left\{ e_1, e_2, e_5\right\}$: $A$ contains the three elementary events $e_1$, $e_2$, and $e_5$. 
		
	\end{alertblock}
	
	\begin{alertblock}{Definition: Event Set}
		\begin{myframe}{
				All events of a random experiment with sample space $\Omega$ form the corresponding \textbf{event set} ${E(\Omega)}$.
				%\marginpar{{\sffamily\mdseries\upshape Grundgesamtheit}}
			}
		\end{myframe}
	\end{alertblock}
	%
	\framebreak
	%
	\begin{alertblock}{Remarks on the Event Set $E(\Omega)$}
		All events means "the set of all subsets of the sample space $\Omega$". In set theory, this is the power set of a base set (here, the sample space $\Omega$)\footnote{Also denoted by $\mathcal P(\Omega)$}, i.e., a set whose elements are themselves sets.
		\par\smallskip
		Note that
		\begin{itemize}
			\item the \textbf{certain event}, the base set $\Omega$, belongs to $E(\Omega)$: $\Omega \subseteq \Omega$. Since it contains all elementary events, it always occurs whenever the random experiment is performed.
			\item the \textbf{impossible event} $\emptyset$ belongs to $E(\Omega)$: $\emptyset \subset \Omega$. Since the empty set has no elements, this event never occurs.
		\end{itemize}
		For finite sample spaces $\Omega$, i.e., $|\Omega| = n \in \mathbb{N}_0$, we have $|E(\Omega)| = 2^n$; in general, $|\Omega| <|E(\Omega)|$ (Cantor's theorem).
		\par\bigskip
		Examples:
		\begin{itemize}
			\item $\Omega = \emptyset \Rightarrow E(\Omega) = \{ \emptyset \}$
			\item $\Omega = \{ e \} \Rightarrow E(\Omega) = \bigl\{ \emptyset, \{ e \}\bigr\}$
			\item $\Omega = \{ e_1, e_2 \} \Rightarrow E(\Omega) = \bigl\{ \emptyset, \{ e_1 \}, \{ e_2 \}, \{ e_1, e_2 \} \bigr\}$
			\item $\Omega = \{ e_1, e_2, e_3 \}$ \par $\Rightarrow E(\Omega) = \bigl\{ \emptyset, \{ e_1 \}, \{ e_2 \}, \{ e_3 \}, \{ e_1, e_2 \}, \{ e_1, e_3 \}, \{ e_2, e_3 \}, \{ e_1, e_2, e_3 \}\bigr\}$
		\end{itemize}
	\end{alertblock}

\subsection{Operations with Events}\label{ch:8.2}
 
\begin{frame}[allowframebreaks,allowdisplaybreaks]
	\frametitle<presentation>{Operations with Events}
	Events are handled mathematically like sets:
	%
	\begin{columns}[T,onlytextwidth]
		\begin{column}{0.7\linewidth}
			\begin{itemize}
				\item \textbf{Union} $A \cup B$: The event \glqq\textbf{\textit{A} or \textit{B}}\grqq\,occurs exactly when $A$, or $B$, or both occur.
				\rule[-0.6cm]{0mm}{0.6cm}\par\smallskip
				\item \textbf{Intersection} $A \cap B$: The event \glqq\textbf{\textit{A} and \textit{B}}\grqq\,occurs exactly when $A$ and $B$ occur simultaneously.
				\rule[-0.9cm]{0mm}{0.9cm}\par\smallskip
				\item \textbf{Difference} $A \backslash B$: The event \glqq\textbf{\textit{A} without \textit{B}}\grqq\,occurs exactly when $A$ occurs but $B$ does not.
				\rule[-0.9cm]{0mm}{0.9cm}\par\smallskip
				\item \textbf{Complement} $\overline{A}$: The event \glqq\textbf{not \textit{A}}\grqq\,occurs exactly when $A$ does not occur.
			\end{itemize}
		\end{column}
		\begin{column}{0.25\linewidth}

	\begin{alertblock}{Definition: Disjoint Events}
		\begin{myframe}{
				Two events $A$ and $B$ are called \textbf{disjoint} or \textbf{mutually exclusive} if
				\begin{equation*}
					\label{eq:8-2}
					A \cap  B = \emptyset
				\end{equation*}
			}
		\end{myframe}
	\end{alertblock}
	%
	\begin{alertblock}{Definition: Complementary Event}
		\begin{myframe}{
				The event 
				\begin{equation*}
					\label{eq:8-3}
					\overline{A} = \Omega \backslash A
				\end{equation*}
				is called the \textbf{complementary event} of $A$.
			}
		\end{myframe}
	\end{alertblock}
	%
	\begin{alertblock}{Definition: Implied Event}
		\begin{myframe}{
				If event $A$ always occurs whenever event $B$ occurs, i.e., $A \subset B$, then we say that \textbf{event \textit{B} implies event \textit{A}}.
			}
		\end{myframe}
	\end{alertblock}
	%
	\par\medskip
	\alert{Note:} The cardinality $|A|$ of a set is the number of elements contained in $A$.
	%
\end{frame}


\subsection{Probability}
%
\begin{frame}%[allowframebreaks,allowdisplaybreaks]
	\frametitle<presentation>{Probability}
	The possible events of a random experiment each have a certain chance of occurring. 
	\par\medskip
	It is therefore natural to assign a real number to every element of the event set $E(\Omega)$ that \alert{quantifies} this chance. 
	\par\medskip
	Such a mapping $P$ is a real-valued function given by
	%
	\begin{eqnarray*}
		P \; : \; E(\Omega) &\to & \MRn  \\
		A &\mapsto & P(A)
	\end{eqnarray*}  
	This defines the probability $P(A)$ of an event $A$ in a formal way, but how exactly can probability be measured or specified, and which rules does it obey?
	\par\bigskip
	The answer is not entirely simple; there are
	\begin{enumerate}
		\item the \glqq classical\grqq\, concept of probability,
		\item the \glqq axiomatic\grqq\, concept of probability.
	\end{enumerate}
	%
	\par\bigskip

\begin{frame}%[allowframebreaks,allowdisplaybreaks]
	\frametitle<presentation>{Axioms of Probability Theory}
	%
	\begin{alertblock}{Definition: Axioms of Probability Theory}
		\begin{myframe}{
			%\small
			Any function $P$
			\begin{eqnarray*}
				P \; : \; E(\Omega) &\to & \MRn  \\
				A &\mapsto & P(A)
			\end{eqnarray*} 
			that assigns a real number to each event
				$A$ from the event set $E(\Omega)$ is called a \textbf{probability function}, and the assigned real
				number $P(A)$ is called the \textbf{probability} of $A$, provided that the following three
				Kolmogorov\footnote{Andrej Nikolaevi\v{c} Kolmogorov, 1903--1987} axioms are satisfied:
				\vspace{0.2em}
				%\begin{enumerate}
				%\item $P(A) \geq 0$ für jedes $A \in E$ \newline
				%Die Wahrscheinlichkeit $P(A)$ jedes Ereignisses $A$ ist eine nichtnegative relle Zahl.\newline
				%\item $P(\Omega) = 1$ \newline
				%Das sichere Ereignis $\Omega$ hat die Wahrscheinlichkeit 1.\newline
				%\item $P(A \cup B) = P(A) + P(B)$ falls $A \cap B = \emptyset$\newline
				%Additionsregel für disjunkte Ereignisse.
				%\end{enumerate}
				\begin{description}
					\setlength{\itemsep}{0.25em}
					\item[\textbf{Axiom K1:}] $P(A) \geq 0 \quad \forall A \subset E$\\
					The probability $P(A)$ of every event $A$ is a non-negative real number.
					\vspace{3mm}
					\item[\textbf{Axiom K2:}] $P(\Omega) = 1$\\
					The certain event $\Omega$ has probability 1.
					\vspace{3mm}
					\item[\textbf{Axiom K3:}] $P(A \cup B) = P(A) + P(B)$ if $A \cap B = \emptyset$\\
					Addition rule for disjoint events.
				\end{description}
			}
		\end{myframe}
	\end{alertblock}
	%
	The tuple $[\Omega, E, P]$ is called a probability space.

\subsection{Rules of Probability}
\begin{frame}[allowframebreaks,allowdisplaybreaks]
	\frametitle<presentation>{Rules of Probability}
	%
	Kolmogorov's axioms only specify the rules that a probability measure must satisfy.
	%
	\begin{alertblock}{Theorem 1: Complementary Event} \label{theo:1}
		The probability of the event complementary to $A$ is always
		\begin{equation*}
			P(\overline{A}) = 1 - P(A), \quad \forall A \subset E
		\end{equation*}
	\end{alertblock}
	%
	\begin{alertblock}{Theorem 2: Impossible Event} \label{theo:2}
		The impossible event has probability
		\begin{equation*}
			P(\emptyset) = 0
		\end{equation*}
	\end{alertblock}
	%
	\begin{alertblock}{Theorem 3: Pairwise Disjoint Events} \label{theo:3}
		If the events $A_1, A_2, \ldots, A_n \; \subset\; E$ are pairwise disjoint, then the probability of the event formed by the union of all these events equals the sum of the individual probabilities:
		\begin{equation*}
			P(A_1 \cup A_2 \cup \ldots \cup A_n) = \sum_{j=1}^{n} P(A_j)
		\end{equation*}
	\end{alertblock}
	%
	\begin{alertblock}{Theorem 4: Difference Events} \label{theo:4}
		For a difference set $A \backslash B$, the following always holds:
		\begin{equation*}
			P(A \backslash B) = P(A) - P(A \cap B)
		\end{equation*}
	\end{alertblock}
	%
	\begin{alertblock}{Theorem 5: Addition Rule} \label{theo:5}
		For any two events $A$ and $B$ from $E$, the following always holds:
		\begin{equation*}
			P(A \cup B) = P(A) + P(B) -  P(A \cap B)
		\end{equation*}
	\end{alertblock}
	%
	\begin{alertblock}{Theorem 6: Monotonicity Property} \label{theo:6}
		If event $A$ implies event $B$, then the probability of $B$ is never smaller than the probability of $A$:
		\begin{equation*}
			A \subset B \Rightarrow P(A) \leq P(B)
		\end{equation*}
	\end{alertblock}
	%
\end{frame}

\subsection{Conditional Probability \& Stochastic Independence}\label{ch:8.9}
%%
\begin{frame}[allowframebreaks,allowdisplaybreaks]
	\frametitle<presentation>{Conditional Probability \& Stochastic Independence}
	%
	The probability of occurrence of an event $A$ "`under the condition"' that event $B$ has occurred (or occurs simultaneously with $A$) is called the "`conditional probability of $A$ given $B$"' and is written as $P(A|B)$.
	\par\medskip
	%
	\begin{alertblock}{Definition: Conditional Probability}
		\begin{myframe}{
				Let $A$ and $B$ be two events of the probability space $[\Omega, E, P]$. The \textbf{conditional probability of \textit{A} given \textit{B}} is defined as
				%\setcounter{equation}{14}
				\begin{equation*}
					\label{eq:8-15}
					P(A|B) = \dfrac{P(A\cap B)}{P(B)}, \quad P(B) > 0
				\end{equation*}
				%\marginpar{{\sffamily\mdseries\upshape Grundgesamtheit}}
			}
		\end{myframe}
	\end{alertblock}
	\par\medskip
	\begin{alertblock}{Theorem 7: Multiplication Rule} \label{theo:7}
		The probability that two events $A$ and $B$ occur simultaneously is
		\begin{equation*}
			\label{eq:8-16}
			P(A\cap B) = P(A|B) \cdot  P(B) = P(B|A) \cdot  P(A) = P(B\cap A)
		\end{equation*}
	\end{alertblock}
	
	\framebreak
	
	\textbf{Experiment:}  
	A red and a green die are rolled (36 elementary events):
	\begin{center}
	\includegraphics[width=0.45\textwidth]{Bilder/Conditional_Probability_Ex1.png}
	\end{center}
	
	%\vspace{0.1cm}
	
	\textbf{Events:}
	\begin{itemize}
		\item $A$: ``At least one die shows a $6$''
		\item $B$: ``Sum of the dice/pips is greater than $9$''
	\end{itemize}
	
	\vspace{0.1cm}
	
	\textbf{Probabilities:}
	$$P(A) = \frac{11}{36}, \qquad P(B) = \frac{6}{36}, \qquad P(A \cap B) = \frac{5}{36}$$
%	\begin{itemize}
%		\item $P(A) = \frac{11}{36}$
%		\item $P(B) = \frac{6}{36}$
%		\item $P(A \cap B) = \frac{5}{36}$
%	\end{itemize}
	
	%\vspace{0.1cm}
	
	\textbf{Conditional Probability:}
	$$P(A \mid B) = \frac{P(A \cap B)}{P(B)}
	= \frac{5/36}{6/36}
	= \frac{5}{6}$$
	
	%\vspace{0.1cm}
	
	\textbf{Interpretation:}  
	Given that the sum exceeds 9, the probability that at least one die shows a six is $\frac{5}{6}$.

	\begin{subequations}
		\begin{alertblock}{Definition: Stochastic Independence}
			\begin{myframe}{
					Two events $A$ and $B$ are called \textbf{(stochastically) independent} if
					\begin{eqnarray*}
						\label{eq:8-17}
						P(A|B) &=& P(A) \\
						P(B|A) &=& P(B)
					\end{eqnarray*}
					%\marginpar{{\sffamily\mdseries\upshape Grundgesamtheit}}
				}
			\end{myframe}
		\end{alertblock}
	\end{subequations}
	%
	\begin{alertblock}{Theorem 8: Multiplication Rule for Independent Events} \label{theo:8}
		If two events $A$ and $B$ are independent, then the probability that $A$ and $B$ occur simultaneously is exactly the product of the individual probabilities:
		\begin{equation*}
			\label{eq:8-18}
			P(A\cap B) = P(A) \cdot  P(B) 
		\end{equation*}
	\end{alertblock}
	%%
\end{frame}

\begin{frame}%[allowframebreaks,allowdisplaybreaks]
	\frametitle<presentation>{Law of Total Probability}
	%
	\begin{alertblock}{Definition: Partition}
		\begin{myframe}{
				Any $n$ disjoint events $H_1 \subset \Omega, H_2\subset \Omega, \ldots, H_n \subset \Omega$ that together cover the event space $\Omega$ completely,
				\begin{subequations}
					\begin{align*}
						H_i \cap H_j &= \emptyset\quad \forall\; i \neq j \\ %\label{eq:8-19a}\\
						H_1 \cup H_2 \cup \ldots \cup H_n &= \Omega %\label{eq:8-19b}
					\end{align*}
				\end{subequations}
				are called a \textbf{partition} of $\Omega$.
				%\marginpar{{\sffamily\mdseries\upshape Grundgesamtheit}}
			}
		\end{myframe}
	\end{alertblock}
	%
	%\par
	\begin{alertblock}{Theorem 9: Law of Total Probability}
		Let $H_1, H_2, \ldots, H_n$ be a partition of $\Omega$. Then for every $B \subset E$,
		\begin{equation*}
			\label{eq:8-20}
			P(B) = \sum_{j=1}^{n} P(B|H_j) \cdot P(H_j) 
			\includegraphics[trim=-2cm 1.5cm 0cm 0cm, width=0.4\linewidth]{Bilder/Teil-II-Kapitel-5-Bild-04.jpg}
		\end{equation*} 
	\end{alertblock}
	%
	\begin{alertblock}{Note}
		The term "`total probability"' expresses that the probability of an event can be determined based on an arbitrarily chosen partition, provided that the conditional probabilities and the probabilities of the conditions are known.
	\end{alertblock}
\end{frame}

\begin{frame}{Example: Manufacturing}
	
	\textbf{Problem:}  
	A product is manufactured by two machines:
	\begin{itemize}
		\item Machine $M_1$: produces $\frac{2}{3}$ of all items, defect rate $10\%$
		\item Machine $M_2$: produces $\frac{1}{3}$ of all items, defect rate $7\%$
	\end{itemize}
	
	\vspace{0.3cm}
	
	\textbf{Given:}
	\begin{itemize}
		\item $P(\text{defect} \mid M_1) = 0.10$, \quad $P(M_1) = \frac{2}{3}$
		\item $P(\text{defect} \mid M_2) = 0.07$, \quad $P(M_2) = \frac{1}{3}$
	\end{itemize}
	
	\vspace{0.3cm}
	
	\textbf{Law of Total Probability:}
	\[
	P(\text{defect}) = P(\text{defect} \mid M_1) \cdot P(M_1)
	+ P(\text{defect} \mid M_2) \cdot P(M_2)
	\]
	
	\vspace{0.3cm}
	
	\textbf{Calculation:}
	\[
	P(\text{defect}) = 0.10 \cdot \frac{2}{3}
	+ 0.07 \cdot \frac{1}{3}
	= 0.09
	\]
	
	\vspace{0.3cm}
	
	\textbf{Interpretation:}  
	The overall probability that a randomly selected item is defective is $9\%$.
	
\end{frame}

\begin{frame}[allowframebreaks,allowdisplaybreaks]
	\frametitle<presentation>{Example: Three-Door Problem or Monty Hall Problem}
	
	\textbf{The problem:}
	\begin{itemize}
		\item Behind one of three doors there is a car; behind each of the others there is a goat.
		\item Hoping to win the car, the contestant first chooses one door.
		\item The host, who knows what is behind the doors, then opens another door with a goat behind it and offers the contestant the option to switch.
	\end{itemize}
	\vspace{0.2cm}
	\begin{center}
		\begin{figure}
			\includegraphics[width=0.6\linewidth]{Bilder/monty_open_door.png}\caption{Wikipedia, \emph{Monty Hall problem}, accessed on 02.05.2024}
		\end{figure}
	\end{center}
	%\vspace{0.2cm}
	\textbf{Question:} Is it advantageous for the contestant to switch doors?
	
	\framebreak
	
	The door-choice events are defined as follows:
	\begin{itemize}
		\item $g_1$: The contestant \emph{initially} chose the door with the car
		\item $g_2$: The contestant \emph{initially} chose a door with a goat
		\item $G_1$: The contestant wins the car \emph{after switching}
		\item $G_2$: The contestant wins a goat \emph{after switching}
	\end{itemize}
	\vspace{0.3cm}
	Initial event probabilities (without switching): $$P(g_1)=\frac 13 \;\text{ and }\; P(g_2)=\frac 23$$
	
	\vspace{0.3cm}
	Conditional event probabilities when switching: $$P(G_1|g_1) = 0 \;\text{ and }\; P(G_1|g_2) = 1$$
	
	\vspace{0.3cm}
	Event probability when switching: $$P(G_1) = P(G_1|g_1)P(g_1) + P(G_1|g_2)P(g_2) = 0\cdot\frac 13 + 1 \cdot \frac 23 = \frac 23$$
	
\end{frame}


\subsection{Bayes' Rule}\label{ch:8.11}
%
\begin{frame}[allowframebreaks,allowdisplaybreaks]
	\frametitle<presentation>{Bayes' Rule}
	%
	Bayes' rule establishes a relationship between two conditional probabilities in which the roles of condition and event are interchanged, e.g.,
	\begin{equation*}
		P(A|B) \text{ and } P(B|A)
	\end{equation*}
	%
	By rearranging Theorem 7, we obtain
	\begin{equation*}
		\dfrac{P(A|B)}{P(B|A)} =  \dfrac{P(A)}{P(B)}
	\end{equation*}
	%
	and therefore
	\begin{equation*}
		P(A|B) = \dfrac{P(A)}{P(B)}\cdot P(B|A) = \dfrac{P(B|A) \cdot P(A)}{P(B)} = \dfrac{P(B \cap A)}{P(B)}
	\end{equation*}
	%
	\begin{alertblock}{Note}
		The above relationship is often interpreted as follows: 
		\begin{itemize}
			\item $A$ is an event whose probability $P(A)$ is known before ("`\alert{a priori}"') carrying out the random experiment. 
			\item $B$ is an event that occurs in the outcome with probability $P(B)$. 
			\item The relation above quantifies how the probability of $A$ changes afterwards ("`\alert{a posteriori}"') when it is known that $B$ has actually occurred, i.e., $P(A) \mapsto P(A|B)$.
	\end{itemize}
	\end{alertblock}


	Bayes' rule also applies to an element $H_i$ of a partition, and together with Theorem 9 this yields
	%
	\begin{equation*}
		P(H_i|B) =  \dfrac{P(H_i) \cdot P(B|H_i)}{P(B)} = \dfrac{P(H_i) \cdot P(B|H_i)}{\displaystyle \sum_{j=1}^{n} P(B|H_j) \cdot P(H_j)}
	\end{equation*}
	from which Bayes' theorem follows directly:
	\par\bigskip
	%
%	\alert{Theorem 10: Bayes' Theorem}
%	\newline
	\begin{alertblock}{Bayes' Theorem}
		\begin{myframe}{
	Let $H_1, H_2, \ldots, H_n$ be an arbitrary partition of $\Omega$ and let $B$ be an event with $P(B) > 0$. Then
	\begin{equation*}
		\label{eq:8-21}
		P(H_i|B) =  \dfrac{ P(B|H_i) \cdot P(H_i) }{\displaystyle \sum_{j=1}^{n} P(B|H_j) \cdot P(H_j)}
	\end{equation*}
        }\end{myframe}
    \end{alertblock} 
	%
\end{frame}

\begin{frame}{Example: Manufacturing}
	
	\textbf{Context (from previous example):} \\ 
	A product is manufactured by two machines:
	\begin{itemize}
		\item $M_1$: produces $\frac{2}{3}$ of items, defect rate $10\%$
		\item $M_2$: produces $\frac{1}{3}$ of items, defect rate $7\%$
	\end{itemize}
	
	\vspace{0.3cm}
	
	\textbf{Question:}  
	Given that a randomly selected item is defective, what is the probability that it was produced by $M_1$?
	
	\vspace{0.3cm}
	
	\textbf{Recap:}
	$$P(\text{defect}) = 0.09$$
	
	\vspace{0.3cm}
	
	\textbf{Bayes' Theorem:}
	$$P(M_1 \mid \text{defect}) =
	\frac{P(\text{defect} \mid M_1)\,P(M_1)}
	{P(\text{defect})} =
	\frac{0.10 \cdot \frac{2}{3}}{0.09}
	= \frac{20}{27}
	\approx 0.741$$
	
	\vspace{0.3cm}
	
	\textbf{Interpretation:}  
	If a product is defective, the probability that it was produced by $M_1$ increases from $66.7\%$ to about $74.1\%$.
	
\end{frame}

\begin{frame}
	\frametitle<presentation>{Example: Three-Door Problem or Monty Hall Problem}
	
	\alert{The following events are defined:}
	\begin{itemize}
		\item $G_i$: The car is behind door $i$ ($i=1,2,3$)
		\item $M_j$: The host opened door $j$ ($j=1,2,3$)
	\end{itemize}
	\vspace{0.3cm}
	\alert{The following situation is given:}\\ 
	The contestant chose door 1, and the host then opened door 3. \linebreak
	$\Longrightarrow$ We are looking for the \emph{conditional probability} $P(G_{2}|M_{3})$. % (that the car is behind door 2 given that it is not behind door 3.)
	
	\vspace{0.3cm}
	\alert{The following probabilities are obtained:}
	\begin{itemize}
		\item $P(G_1) = P(G_2) = P(G_3) = \frac 13$
		\item $P(M_3|G_1) = \frac 12$, $P(M_3|G_2) = 1$, and $P(M_3|G_3) = 0$
	\end{itemize}
	
	\vspace{0.3cm}
	\alert{Applying \emph{Bayes' theorem} yields:} 
	\begin{align*}
		P(G_2|M_3) & = \frac{P(M_3|G_2)P(G_2)}{P(M_3|G_1)P(G_1)+P(M_3|G_2)P(G_2)+P(M_3|G_3)P(G_3)} \\
		& = \frac{1\cdot\frac 13}{\frac 12\cdot\frac 13+1\cdot\frac 13+0\cdot\frac 13} = \frac{\frac 13}{\frac 12} = \frac 23
	\end{align*}
	
	\vspace{0.1cm}
	\alert{Consequence:} The contestant should switch in order to double the winning probability from initially $\frac 13$ to now $\frac 23$.
\end{frame}

\subsection{Definition}
% ------------------------------------------------------------
\begin{frame}%[allowframebreaks,allowdisplaybreaks]
	\frametitle<presentation>{Definition of Random Variables}
	%
	\begin{itemize}
		\item Until now: Random events to which probabilities were assigned.
		\item From now on: Assign probabilities to real numbers: If an elementary event $e \subset \Omega$ occurs, then a real number $\ZVX(e)$ is assigned to this $e$.
	\end{itemize}
	%
	\begin{alertblock}{Definition: Random Variable}
		\begin{myframe}{
				Given a probability space $[\Omega, E, P]$. A function
				\begin{eqnarray*}
					\ZVX \; : \; \Omega &\to     & \MRn    \nonumber     \\
					e &\mapsto & \ZVX(e) \label{eq:9-1}
				\end{eqnarray*}  
				that assigns a real number $\ZVX(e)$ to each elementary event $e$ is called a \textbf{random variable} or \textbf{stochastic variable} if, for every real number $x$, there exists an event $A_x \subset E$ with $A_x = \{e|\ZVX(e) \leq x \}$.
			}
		\end{myframe}
	\end{alertblock}
	%
	\alert{Remarks:}
	%
	\begin{itemize}
		\item The random \textit{variable} $\ZVX$ is a \textit{function}.
		\item The probability function $P$ is not needed for the definition.
		\item Introducing random variables largely eliminates the rather exhausting calculations with elementary events and events.
	\end{itemize}
	%
\end{frame}

\subsection{Distribution Function}
\begin{frame}[allowframebreaks,allowdisplaybreaks]
	\frametitle<presentation>{Distribution Function}
	%
	In addition to the real numbers $x$ that a random variable $\ZVX$ can assume, the probabilities $P$ with which these values occur are also of interest. Let
	%
	\begin{align*}
		P(\ZVX = a)    & \Leftrightarrow\text{probability that }\ZVX = a \\
		P(a <\ZVX \leq b) & \Leftrightarrow\text{probability that }\ZVX \text{ is greater than } a \text{ and at most } b\\
		P(\ZVX \leq x) & \Leftrightarrow\text{probability that }\ZVX \leq x
	\end{align*}
	%
	Each of the three cases above can be uniquely associated with events and, through them, with the corresponding probabilities:
	\begin{equation*}
		\begin{array}{rcl}
			\ZVX = a    & \Leftrightarrow & A = \{e | \ZVX(e) = a \} \\
			P(\ZVX = a) & =               & P\left( A \right) = P\left( \{e | \ZVX(e) = a \}\right) 
		\end{array}
	\end{equation*}
	Accordingly,
	\begin{equation*}
		\begin{array}{rcl}
			a <\ZVX \leq b    & \Leftrightarrow & A = \{e | a <\ZVX(e) \leq b \} \\
			P(a <\ZVX \leq b) & =               & P\left( A \right) = P\left( \{e | a <\ZVX(e) \leq b \}\right) 
		\end{array}
	\end{equation*}
	Finally, for every real number $x$,
	\begin{equation*}
		\begin{array}{rcl}
			\ZVX \leq x    & \Leftrightarrow & A_x = \{e | \ZVX(e) \leq x \} \\
			P(\ZVX \leq x) & =               & P\left( A_x \right) = P\left( \{e | \ZVX(e) \leq x \}\right) 
		\end{array}
	\end{equation*}
	%
	\framebreak
	\begin{alertblock}{Definition: Distribution Function}
		\begin{myframe}{
				The function
				\begin{equation*}
					\label{eq:9-2}
					F(x) = P(\ZVX \leq x) = P\left( A_x \right) = P\left( \{e | \ZVX(e) \leq x \}\right) 
				\end{equation*}  
				which assigns to each real number $x$ the probability that the random variable $\ZVX$ takes on a value $\ZVX \leq x$, is called the \textbf{distribution function} of the random variable $\ZVX$. This probability is equal to the probability of the event $A_x = \{e | \ZVX(e) \leq x \}$.
				%\marginpar{{\sffamily\mdseries\upshape Grundgesamtheit}}
			}
		\end{myframe}
	\end{alertblock}
	%
	\par\medskip
	The distribution function has the following properties:
	%
	\begin{enumerate}
		\item $F(x)$ is at least right-continuous at every point:
		\begin{equation*}  
			\limes_{\Delta x \rightarrow 0} F(x + \Delta x) = F(x), \quad \Delta x > 0
		\end{equation*}
		\item $F(x)$ is monotonically increasing everywhere:
		\begin{equation*}  
			F(a) \leq F(b) , \quad \text{if } a < b, \quad a,b \in \MRn
		\end{equation*}  
		\item $F(x)$ has the limits
		\begin{equation*}  
			\limes_{x \rightarrow -\infty} F(x) = 0, \quad \limes_{x \rightarrow +\infty} F(x) = 1
		\end{equation*}
	\end{enumerate}

	\par\smallskip
	\alert{Note}
	\par\smallskip
	The properties of the distribution function are identical to those of the empirical distribution function!

\subsection{Discrete Random Variables}
%
\begin{frame}[allowframebreaks,allowdisplaybreaks]
	\frametitle<presentation>{Probability Mass Function}
	In finite probability spaces, i.e.\ $|\Omega| < \infty$, the random variable is discrete, and there are $n$ distinguishable real numbers $x_i$, $i=1,\dots,n$ as realizations of $\ZVX$.
	%
	\par\medskip
	%
	\begin{alertblock}{Definition: Probability Mass Function}
		\begin{myframe}{
				If $\ZVX$ is a discrete random variable, then the function
				\begin{equation*}
					\label{eq:9-3}
					f(x) = P(\ZVX = x) =  
					\begin{cases}
						P(\ZVX = x_i) = p_i &\;\text{if } x = x_i,\; i=1,\ldots, n \\
						0                   & \;\text{otherwise}
					\end{cases}
				\end{equation*}  
				is called the \textbf{probability mass function} or simply the \textbf{mass function} of the random variable $\ZVX$ (sometimes also called the probability function).
				%\marginpar{{\sffamily\mdseries\upshape Grundgesamtheit}}
			}
		\end{myframe}
	\end{alertblock}
	%
	%
	%\par\medskip
	%
	\begin{columns}
		\begin{column}{0.27\linewidth}
			\alert{Properties} 
			\begin{itemize}
				\item $f(x) \geq 0$\label{eq:9-4a}
				\item $\sum\limits_{\text{all }i} f(x_i) = 1$\label{eq:9-4b}
				\item $0 < f(x_i) \leq 1$\label{eq:9-4c}
			\end{itemize}
		\end{column}
		%
		\hfill
		\begin{column}{0.64\linewidth}
			\alert{Relationships} 
			%\par\medskip
			%
			\begin{equation*}
				\label{eq:9-5}
				F(x) = P(\ZVX \leq x) = \sum_{x_i \leq x} f(x_i)
			\end{equation*}
			\begin{equation*}
				\label{eq:9-6}
				\begin{aligned}
					f(x) &= F(x) - \limes_{\Delta x \rightarrow 0} F(x - \Delta x),
					&\Delta x > 0
				\end{aligned}
			\end{equation*}
		\end{column}
		%
	\end{columns}
	% 
	\begin{alertblock}{Note} 
		The distribution function $F(x)$ of discrete random variables is not differentiable at the jump points $x = x_i$!
	\end{alertblock}

\subsection{Continuous Random Variables}

\begin{frame}[allowframebreaks,allowdisplaybreaks]
	\frametitle<presentation>{Density Function}
	In infinite probability spaces, the random variable is continuous. The realizations of $\ZVX$ typically lie in intervals of the real numbers, for example in the interval $[a,b], \; a{,}b \in \MRn$. 
%	Since such an interval contains infinitely many real numbers, the probability of hitting exactly one specific number is always zero. Therefore, specifying a probability mass function is not meaningful.
	%
	\par\medskip
	%
	\begin{alertblock}{Definition: Probability Density Function}
		\begin{myframe}{
				If $\ZVX$ is a continuous random variable with distribution function $F(x)$, then the first derivative of $F(x)$ with respect to $x$
				\begin{equation*}
					\label{eq:9-7}
					f(x) = \dfrac{dF(x)}{dx}
				\end{equation*}  
				is called the \textbf{probability density function}, or simply the \textbf{density function}, of the random variable $\ZVX$.
				%\marginpar{{\sffamily\mdseries\upshape Grundgesamtheit}}
			}
		\end{myframe}
	\end{alertblock}
	
	\par\medskip
	%
	\begin{columns}
		\begin{column}{0.40\linewidth}
			\alert{Properties} 
			\begin{subequations} 
				\begin{eqnarray*}  
					(1) & f(x) \geq 0 \label{eq:9-8a}\\
					(2) & \displaystyle \int\limits_{-\infty}^{+\infty} f(x) \operatorname{d}x = 1 \label{eq:9-8b}
				\end{eqnarray*}  
			\end{subequations}
		\end{column}
		%
		%\hfill
		\begin{column}{0.55\linewidth}
			\alert{Note}\par\smallskip 
			In the case of continuous random variables, $F(x)$ is in general everywhere differentiable, and, hence, $f(x)$ is everywhere defined.

		%
	\end{columns}

			P(\ZVX \leq x) & = \int_{-\infty}^{x} f(u) \operatorname{d}u 
		\end{align*}
		%
		\item The interval $[a,b]$ can be decomposed into three disjoint intervals: $[a] \cup (a,b) \cup [b]$. For a finite continuous $f(x)$, the following holds:
		%\begin{align*}
		%%P(\ZVX = c)    & = \int_{c}^{c} f(x) \operatorname{d}x = 0 \\

\subsection{Expected Value}
%
\begin{frame}[allowframebreaks,allowdisplaybreaks]
	\frametitle<presentation>{Expected Value of a Random Variable}
	%
	The expected value is comparable to the empirical arithmetic mean.
	It describes the number that the random variable $\ZVX$ takes \glqq on average\grqq\,(a measure of location) $\rightarrow$ a number that does not necessarily have to be a realization of $\ZVX$.
	%
	\par\smallskip
	\begin{alertblock}{Definition: Expected Value of a Random Variable}
		\begin{myframe}{
				Let $\ZVX$ be a random variable and $f(x)$ its mass or density function. Its \textbf{expected value} is defined as \newline
				\begin{align*} 
						\E(\ZVX) &= \displaystyle \sum_{\text{all }j} x_j \cdot f(x_j)\quad \text{discrete} \\
						\E(\ZVX) &= \displaystyle \int\limits_{-\infty}^{+\infty} x\cdot f(x) \,dx\quad \text{continuous}  
				\end{align*}
			}
		\end{myframe}
	\end{alertblock}
	%
	\begin{alertblock}{Notation:}
		The expected value is also called the \textbf{mean} or \textbf{mathematical expectation}.\\
		As a formula symbol for the resulting number, one usually uses $\mu$, $\mu_{\mathrm{ X}}$, or $\mu(\ZVX)$.
	\end{alertblock}
	
	\framebreak

	\begin{alertblock}{Example A (discrete random variable)}
		Let the \alert{discrete} random variable $\ZVX$ have the following \alert{probability mass function}:
		$$f(x)=
		\begin{cases}
			\dfrac{1}{3}, & \text{for } x=1,\\[4pt]
			\dfrac{2}{3}, & \text{for } x=2,\\[4pt]
			0, & \text{otherwise.}
		\end{cases}$$
		Then the expected value is $$\E(\ZVX)=1\cdot \frac{1}{3} + 2\cdot \frac{2}{3} = \frac{5}{3} \approx 1.666\ldots$$
	\end{alertblock}
	
	\begin{alertblock}{Example B (continuous variable)}
		Let the \alert{continuous} random variable $\ZVX$ have the following \alert{density function}:
		$$f(x)=
		\begin{cases}
			\dfrac{3}{4}, & \text{for } 0 \le x \le 1,\\[4pt]
			0, & \text{otherwise.}
		\end{cases}$$
		Then the expected value is:
		\begin{align*}
			\mu = \E(\ZVX) & =\int_{0}^{1} x\cdot f(x)\,dx
			= \int_{-\infty}^1 0\,dx + \int_1^3 x\cdot \frac 14 x\,dx + \int_3^\infty 0\,dx \\
			& = \int_1^3 \frac 14 x^2\,dx = \left[\frac{1}{12}x^3\right]_1^3 = \frac{27}{12}-\frac{1}{12} = \frac{26}{12} = 2.1667.
		\end{align*}
		
		\begin{center}
            \includegraphics[scale=0.75]{Bilder/ExpectedValue.png}
		\end{center}
	\end{alertblock}
	
\end{frame}

%
%\subsection{Rechnen mit Erwartungswerten} 
\begin{frame}[allowframebreaks,allowdisplaybreaks]
	\frametitle<presentation>{Expected Value of Transformation $\ZVY = g(\ZVX)$}
	%
	%\par\medskip
	%
	Let $\ZVY = g(\ZVX)$ be the transformation of the random variable $\ZVX$.	Then it holds:
	\begin{align*}
		\E\left(\ZVY = g(\ZVX)\right) & = \displaystyle \sum_{\text{all }j} g(x_j) \cdot f(x_j)\quad \text{discrete}\\
		\E\left(\ZVY = g(\ZVX)\right) & = \displaystyle \int\limits_{-\infty}^{+\infty} g(x) \cdot f(x) dx\quad \text{continuous}
	\end{align*}   
	

	The following rules apply for \alert{calculations with expected values}:
	\par\smallskip
	%
	\begin{enumerate}
		\item \alert{Constant:} For $y = g(x) = a$, it follows that
		\begin{equation*}
			\E(\ZVY) = \E(a) = a
		\end{equation*}
		\item \alert{Factor:} For $y = b\cdot g(x)$, it follows that
		\begin{equation*}
			\E(\ZVY) = \E(b\cdot g(\ZVX)) = b \cdot E\left(g(\ZVX)\right)
		\end{equation*}
		\item \alert{Linear Transformation:} For $y = a + b\cdot x$, it follows that
		\begin{equation*}
			\label{eq:9-13}
			\E(\ZVY) = \E(a +b\cdot \ZVX ) = a + b \cdot E\left(\ZVX\right)
		\end{equation*}
		\item \alert{Linear Combination:} For $y = \lambda_1 \cdot g_1(x) + \lambda_2 \cdot g_2(x)$, it follows that
		\begin{equation*}
			\label{eq:9-12}
			\E(\ZVY) = \E(\lambda_1 \cdot g_1(\ZVX)+ \lambda_2 \cdot g_2(\ZVX)) = \lambda_1 \cdot  E\left(g_1(\ZVX)\right) + \lambda_2 \cdot E\left(g_2(\ZVX)\right)
		\end{equation*}
	\end{enumerate} 
	
	\framebreak    
	
	\textbf{Exemplary Problem:}
    Failures occur during the operation of a production facility. Based on past observations, the following probability distribution is assumed for the number of failures per day $\ZVX$:
	$$\begin{array}{c|cccc}
			x & 0 & 1 & 2 & 3 \\
			\hline
			f(x)=P(\ZVX=x) & 0.35 & 0.40 & 0.15 & 0.10
		\end{array}$$
	
	\vspace{0.1cm}
		
    Resolving failures follow the degressive cost function (in thousand Euro):
	$$g(x) = 5 - \frac{4}{x+1}$$
	
	\vspace{0.1cm}
	
	\textbf{Substituting the expectation of failures}, 
	$\E(\ZVX)=1$ into the cost function yields expected costs of $g\left(\E(\ZVX)\right)=5-\frac 42 = 3$. $\longrightarrow$ daily costs of $3000$ Euro
	
	\vspace{0.1cm}
	
	\textbf{Compute transformed values}
	$$\begin{array}{c|cccc}
		x & 0 & 1 & 2 & 3 \\
		\hline
		g(x) & 1 & 3 & 3.66 & 4
	\end{array}$$
	
	%\vspace{0.1cm}
	
	\textbf{Compute expectation}
	$$\E(g(\ZVX)) = \sum_x g(x)\cdot f(x)
	= 1\cdot0.35 + 3\cdot0.40 + 3.66\cdot0.15 + 4\cdot0.10 = 2.5$$
	
	%\vspace{0.1cm}
	
	\textbf{Conclusion:} \\
    Lower expected daily costs of $2500$ Euro since 
	$\E(g(\ZVX)) \neq g(\E(\ZVX))$.
	
\end{frame}


\subsection{Variance} 
\begin{frame}[allowframebreaks,allowdisplaybreaks]
	\frametitle<presentation>{Variance of a Random Variable}
	The variance corresponds to the empirical variance and represents a \emph{measure of dispersion around the mean}.
	%
	\begin{alertblock}{Definition: Variance of the Random Variable}
		\begin{myframe}{
				Let $\ZVX$ be a random variable and $\mu_{\mathrm{ X}}$ its expected value. The expected value of the squared deviations of the random variable from its expected value
				\begin{equation*}
					\label{eq:9-14}
					\Var(\ZVX) = \E\left( (\ZVX -  \mu_{\mathrm{ X}})^2 \right)
				\end{equation*} 
				is called the \textbf{variance} of the random variable $\ZVX$, provided that the sum or the integral exists. The positive square root of the variance
				\begin{equation*}
					\label{eq:9-15}
					\sigma_{\mathrm{ X}} = +\sqrt{\Var(\ZVX)}
				\end{equation*} 
				is called the \textbf{standard deviation}. 
				%\marginpar{{\sffamily\mdseries\upshape Grundgesamtheit}}
			}
		\end{myframe}
	\end{alertblock}
	%
	\begin{alertblock}{Notation:}
		As a formula symbol, especially for the resulting number, $\sigma^2$, $\sigma^2_{\mathrm{ X}}$, or $\sigma^2(\ZVX)$ is usually used for $\Var(\ZVX)$.
	\end{alertblock}
	%
	\begin{alertblock}{Note:}
		The variance is the expected value of the transformation $y = g(x) = (x - \mu_{\mathrm{ X}})^2$.
	\end{alertblock}
	%
	\framebreak
	%
	According to the definition, the variance is calculated as follows:
	\begin{align*}
		\Var\left(\ZVX\right) & = \displaystyle \sum_{\text{all }j} \left(x_j - \mu_{\mathrm{ X}} \right)^2 \cdot f(x_j)\quad \text{discrete} \\
		\Var\left(\ZVX\right) & = \displaystyle \int_{-\infty}^{+\infty} \left(x - \mu_{\mathrm{ X}} \right)^2 \cdot f(x)\,dx\quad \text{continuous}
	\end{align*}  
	
	\alert{Example:} Given $\ZVX$ with density $f(x)=\frac 32\left(x-\frac 12x^2\right)$ in the interval $[0,2]$:
	\begin{center}
        \includegraphics[scale=0.3]{Bilder/Variance.png}
	\end{center}
	
	According to the symmetry, the expected value is $\mu_{\mathrm{ X}}=\E(\ZVX)=1$.\\
	The variance is computed via the definite integral
	$$\Var(\ZVX)= \int_0^2 (x-1)^2 \cdot \frac 32 \left(x-\frac 12x^2\right)\,dx = \ldots = \frac 15 $$ 
	and the standard deviation yields $\sigma_{\mathrm{ X}} = 0.4472$.
	
	
	\framebreak
	The following rules apply for \alert{calculations with variances}:
	\par\smallskip
	\begin{enumerate}
		\item \alert{Simplified Calculation:} The following holds (see item 6 for $d=0$):
		\begin{equation*}
			\label{eq:9-19}
			\Var(\ZVY) = \E\left( \ZVX^2 \right) - \mu^2_{\mathrm{ X}} 
		\end{equation*}
		\item \alert{Constant:} Since a constant quantity does not vary, the following holds:
		\begin{equation*}
			\Var(a) = 0
		\end{equation*}
		\item \alert{Shift:} For $y = x+a$, it follows that
		\begin{equation*}
			\Var(\ZVY) =\Var(\ZVX+a) = \Var(\ZVX)
		\end{equation*}
		\item \alert{Scaling:} For $y = b \cdot x$, it follows that
		\begin{equation*}
			\Var(\ZVY) = \Var(b \cdot \ZVX) = b^2 \cdot \Var(\ZVX)
		\end{equation*}
		\item \alert{Linear Transformation:} For $y = a + b\cdot x$, it follows that
		\begin{equation*}
			\Var(\ZVY) = \Var(a +b\cdot(\ZVX)) = b^2 \cdot \Var(\ZVX) \label{eq:9-17} %\\
			%\sigma_{\mathrm{ Y}} &=& |b|  \cdot \sigma_{\mathrm{ X}} \label{eq:9-18}       
		\end{equation*}
		\item \alert{Steiner's Translation Theorem:} For $d = \text{const.}$, the following holds:
		\begin{equation*}
			\label{eq:9-20}
			\Var(\ZVY) = \E\left( (\ZVX - d)^2 \right) - (\mu_{\mathrm{ X}} - d)^2 
		\end{equation*}
	\end{enumerate}     
	%
\end{frame}

\subsection{Standardization} 
\begin{frame}%[allowframebreaks,allowdisplaybreaks]
	\frametitle<presentation>{Standardization of a Random Variable}
	%
	Every random variable can be transformed in such a way that the resulting random variable has \textbf{zero mean} and its \textbf{variance equals one}.
	\par\smallskip
	For the purpose of comparability, this is often desirable or necessary.
	%
	\begin{alertblock}{Definition: Standardization}
		\begin{myframe}{
				If $\ZVX$ is a random variable with expected value $\mu$ and standard deviation $\sigma> 0$, then the linearly transformed random variable
				%
				\begin{equation*}
					\label{eq:9-21}
					\ZVZ = \dfrac{\ZVX - \mu}{\sigma} = -\dfrac{\mu}{\sigma} + \dfrac{1}{\sigma} \cdot \ZVX = a + b \cdot \ZVX 
				\end{equation*} 
				is called \textbf{standardized}. 
				%\marginpar{{\sffamily\mdseries\upshape Grundgesamtheit}}
			}
		\end{myframe}
	\end{alertblock}
	%
	\begin{alertblock}{Remarks}
		\begin{enumerate}
			\item As required (see calculation rules), the following holds:
			\begin{eqnarray*}
				E(\ZVZ) &=&  a + b \cdot E\left(\ZVX\right) = -\dfrac{\mu}{\sigma} + \dfrac{1}{\sigma} \cdot \mu = 0  \\
				\sqrt{\Var(\ZVZ)} &=& |b|  \cdot \sqrt{\Var(\ZVX)} = \dfrac{1}{\sigma} \cdot \sigma = 1     
			\end{eqnarray*}
			%The random variable $\ZVZ$ formed this way is always zero-mean, $\mu_{\mathrm{ Z}} = 0$, and has a standard deviation of one, $\sigma_{\mathrm{ Z}} = 1$; it is thus normalized (``standardized'').
			\item Intuitively, the linear transformation $\ZVZ = g(\ZVX) = \frac{\ZVX - \mu}{\sigma}$ first represents a \textbf{shift} $\ZVY = \ZVX- \mu$ by $\mu$, and then a \textbf{scaling} $\ZVZ = \frac{\ZVY}{\sigma}$ 
			by the factor $\frac 1\sigma$.
		\end{enumerate}
	\end{alertblock}
	%
\end{frame}


\subsection{Quantiles}\label{ch:9.10}
\begin{frame}[shrink=5]%[allowframebreaks,allowdisplaybreaks]
	\frametitle<presentation>{Quantiles}
	%
	Just as in the case of empirical frequencies, the quantiles of a distribution function $F(x)$ can be defined with the same interpretation.
	
	\begin{alertblock}{Definition: Quantile}
		\begin{myframe}{
				The number $x_{[q]}$ for which simultaneously
				\begin{equation*} \label{eq:9-32}
					P\left(\ZVX \leq x_{[q]} \right) \geq q \quad \text{and} \quad P\left(\ZVX \geq x_{[q]} \right) \geq 1-q
				\end{equation*}
				holds is called the \textbf{$q$-quantile} of the distribution.
			}
		\end{myframe}
	\end{alertblock}
	%
	\par\smallskip
	
	\begin{alertblock}{Remember:}
		For $q=0.5$, $x_{[q]} = x_{\text{Med}}$ is called the \emph{Median} with $x_{\text{Med}} = F^{-1}(0.5)$:

		\end{center}
	\end{alertblock}

\end{frame}

\begin{frame}[allowframebreaks,allowdisplaybreaks]
	\frametitle<presentation>{Uniform Distribution}
	\begin{alertblock}{Model}
		The discrete random variable $\ZVX_{\mathrm{ U}}$ has $m$ different realizations $x_1, x_2, \ldots, x_m$, all of which have the same probability (\alert{\emph{Laplace experiment}}). \\
		\vspace{0.1cm}
		The distances between individual realizations do not need to be equal, although this is often the case.
	\end{alertblock}
	%
	\begin{alertblock}{Uniform Distribution}
		If the realizations of a uniformly distributed random variable are the natural numbers from $1$ to $m$, then:
		\begin{eqnarray*}
			f_{\mathrm{ U}}(x;m) &=& 
			\begin{cases}
				\dfrac{1}{m} &\quad \text{for } x = 1,2,\ldots,m \\
				0            &\quad \text{otherwise}
			\end{cases} \label{eq:11-1} \\
			%
			\E(\ZVX_{\mathrm{ U}}) &=& \dfrac{m +1}{2}  \label{eq:11-2} \\
			\Var(\ZVX_{\mathrm{ U}}) &=& \dfrac{m^2-1}{12} \label{eq:11-3}
		\end{eqnarray*}
	\end{alertblock}
	%
	\begin{alertblock}{Example}
		For $m=6$, we can consider the classical \emph{\alert{Laplace experiment}} \glqq rolling a die\grqq\,for the random variable $\ZVX_{\mathrm{ U}}$: \glqq number of pips\grqq\,with the following probability mass function $f(x)$ and distribution function $F(x)$ [Schira, Statistische Methoden der VWL und BWL, 2021]:
		\begin{center}
			\includegraphics[scale=0.55]{Bilder/UniformDistrib.png}
		\end{center}
	\end{alertblock}
\end{frame}

\begin{frame}[allowframebreaks,allowdisplaybreaks]
	\frametitle<presentation>{Bernoulli Distribution}
	%
	\begin{alertblock}{Model} 
		The discrete random variable $\ZVX_{\mathrm{ Be}}$ has the two realizations $0$ and $1$. The random experiment has only the events $A$ (``success'', the event of interest occurs) and $\overline{A}$ (``failure'', the event of interest does not occur). The probability of $A$ is $P(A) = p$.
	\end{alertblock}
	%
	\begin{alertblock}{Bernoulli Distribution} 
		\begin{align*}
			f_{\mathrm{ Be}}(x;p) &= 
			\begin{cases}
				1-p &\quad \text{for } x = 0 \\
				p &\quad \text{for } x = 1 \\
				0            &\quad \text{otherwise}
			\end{cases}  \\
			\E(\ZVX_{\mathrm{ Be}}) &= p  \\
			\Var(\ZVX_{\mathrm{ Be}}) &= p\cdot(1-p) 
		\end{align*}
	\end{alertblock}
	%
	\begin{alertblock}{Note} 
		The Bernoulli distribution can be interpreted as the so-called indicator function of the set $A \subset E$. Its expected value equals the probability of the event $A$.
	\end{alertblock}
	
	\begin{alertblock}{Example}
		\begin{enumerate}
			\item To be able to leave the starting area in \emph{Ludo}, one must roll at least one six in three attempts (``success'').
			
			The probability of success is $p = \frac{91}{216} \approx 0.4213$. Thus, the expected value is $0.4213$, and the variance is $pq = 0.4213 \cdot 0.5787 = 0.2438$.
			\item For a Bernoulli distrtibuted random variable $\ZVX_{\mathrm{ Be}}$ with $p=\frac 13$, we obtain the following probability mass function $f(x)$ and distribution function $F(x)$ [Schira, Statistische Methoden der VWL und BWL, 2021]:
			\begin{center}
				\includegraphics[scale=0.45]{Bilder/BernoulliDistrib.png}
			\end{center}
		\end{enumerate}
	\end{alertblock}
\end{frame}

%\subsubsection{Binomial-Verteilung}\label{ch:11.3}
\begin{frame}[allowframebreaks,allowdisplaybreaks]
	\frametitle<presentation>{Binomial Distribution}
	%
	\begin{alertblock}{Model} 
		Urn model \alert{with} replacement: $N$ balls, of which $0 \leq S\leq N$ are black. The discrete random variable $\ZVX_{\mathrm{ Bi}}$ gives the number of black balls drawn in $n$ draws with replacement. Therefore, in every draw the probability of drawing a black ball is constant and equals $p=\frac{S}{N}$.
	\end{alertblock}
	%
	\begin{alertblock}{Binomial Distribution} 
		With the binomial coefficient $\binom{n}{x} = \frac{n!}{x! \cdot (n-x)!}$, we have
		\begin{align*}
			f_{\mathrm{ Bi}}(x;p,n) &= 
			\binom{n}{x} \cdot p^x \cdot (1-p)^{n-x} , \quad x=0,1,\ldots,n \\
			\E(\ZVX_{\mathrm{ Bi}}) &= n \cdot p \\
			\Var(\ZVX_{\mathrm{ Bi}}) &= n \cdot p\cdot(1-p) 
		\end{align*}
	\end{alertblock}
	%
	\begin{alertblock}{Remark} 
		A binomially distributed random variable can also be interpreted as the sum of $n$ independent Bernoulli-distributed random variables $\ZVX_{\mathrm{ Be},1}, \dots, \ZVX_{\mathrm{ Be},n}$, i.e., $\ZVX_{\mathrm{ Bi}} = \sum_{i=1}^n \ZVX_{\mathrm{ Be},i}$.
	\end{alertblock}
	%
	\begin{alertblock}{Example (continued)}
		To be able to leave the starting area in \emph{Ludo}, one must roll at least one six in three attempts (``success'').
		\par\bigskip
		We have already stated that the probability of success is $p = \frac{91}{216} \approx 0.4213$. 
		We want to verify this. 
		\par\bigskip
		To do so, we add the probabilities of the three disjoint events
		$$P(\ZVX_{\mathrm{ Bi}} = 1) + P(\ZVX_{\mathrm{ Bi}} = 2) + P(\ZVX_{\mathrm{ Bi}} = 3)$$
		with $p = \frac{1}{6}$. Rolling at least one six means rolling either one, two, or three sixes. 
		\par\bigskip
		We compute:
		$$\binom{3}{1}\frac 16\left(\frac 56\right)^2 +
		\binom{3}{2}\left(\frac 16\right)^2\frac 56 +
		\binom{3}{3}\left(\frac 16\right)^3 = 
		(3\cdot 25+3\cdot 5+1)\frac{1}{216} = \frac{91}{216}.$$
	\end{alertblock}
\end{frame}

\begin{frame}[allowframebreaks,allowdisplaybreaks]
	\frametitle<presentation>{Poisson Distribution}
	\begin{alertblock}{Model}
		Limit case of the discrete binomial distribution for a ``huge namber'' of trials ($n \to \infty$) together with a ``very small'' success probability ($p \to 0$), while $\lambda = n\cdot p = \text{const.} \in \MRn^{+}$.
	\end{alertblock}
	%
	\begin{alertblock}{Poisson Distribution}
		\begin{eqnarray*}
			f_{\mathrm{ Po}}(x;\lambda) &=& \dfrac{\lambda^x}{x!}\cdot \operatorname{e}^{-\lambda}, \quad x=0,1,\ldots,n  \label{eq:11-12} \\
			\E(\ZVX_{\mathrm{ Po}}) &=& \lambda \label{eq:11-13a} \\
			\Var(\ZVX_{\mathrm{ Po}}) &=& \lambda \label{eq:11-13b}
		\end{eqnarray*}
		%
	\end{alertblock}
	\begin{alertblock}{Remark} 
		\begin{itemize}
			\item Under the conditions $n$ ``large'' and $p$ ``small'' with $0 < \lambda = n \cdot p < \infty$, the Poisson distribution is a good approximation of the exact binomial distribution.
			\item This approximation improves with increasing $n$ increases and decreasing $p$.
		\end{itemize}
	\end{alertblock}
	%
	\begin{alertblock}{Example}
		\begin{itemize}
			\item Number of typos per page, or lightning strikes per hectare per year, \ldots
			
			\item For a Poisson distrtibuted random variable $\ZVX_{\mathrm{ Po}}$ with $\lambda = 4$, we obtain the following probability mass function $f(x)$ [Schira, Statistische Methoden der VWL und BWL, 2021]:
			\begin{center}
				\includegraphics[scale=0.55]{Bilder/PoissonDistrib.png}
			\end{center}
		\end{itemize}
	\end{alertblock}
	
	\begin{alertblock}{Example}
	    \textbf{Problem:} \\
				A player repeatedly bets on the number 17 in roulette. \\
				What is the probability that he wins exactly 8 times in 200 games?
				
				\vspace{0.2cm}
				
				\textbf{Modeling:}
				\begin{itemize}
					\item Probability of success in one game: $p = \frac{1}{37}$
					\item Number of trials: $n = 200$
					\item Expected value: $\lambda = np = \frac{200}{37} \approx 5.4054$
				\end{itemize}
				
				\vspace{0.2cm}
				
				\textbf{Poisson Approximation:}
				\[
				P(\ZVX_{\mathrm{ Po}} = 8) = f_{\mathrm{ Po}}(8;5.4054) = \frac{\lambda^8}{8!} e^{-\lambda}
				= \frac{5.4054^8}{8!} e^{-5.4054}
				\approx 0.0812
				\]
				
				\vspace{0.2cm}
				
				\textbf{Comparison (Binomial):}
				\[
				P(\ZVX_{\mathrm{ Bi}} = 8) = f_{\mathrm{ Bi}}\left(8;\frac{1}{37};200\right)=\binom{200}{8}
				\left(\frac{1}{37}\right)^8
				\left(\frac{36}{37}\right)^{192}
				\approx 0.0814
				\]
				
				\vspace{0.2cm}
				
				\textbf{Conclusion:} \\
				The Poisson approximation is very accurate for large $n$ and small $p$.
	\end{alertblock}
\end{frame}

%\subsubsection{Geometrische Verteilung}\label{ch:11.6}
\begin{frame}[allowframebreaks,allowdisplaybreaks]
	\frametitle<presentation>{Geometric Distribution}
	%
	\begin{alertblock}{Model} 
		A Bernoulli experiment with success probability $p$ is repeated independently until the first success occurs. The discrete random variable $\ZVX_{\mathrm{ Geo}} \in \MNn_0$ gives the number of \alert{preceding failures}.
	\end{alertblock}
	%
	\begin{alertblock}{Geometric Distribution}
		\begin{align*}
			f_{\mathrm{ Geo}}(x;p) &= \left( 1-p \right)^x \cdot p, \quad x = 0,1,2,\dots  \\
			F_{\mathrm{ Geo}}(x;p) &= 1-\left(1-p\right)^{x+1} \\
			\E(\ZVX_{\mathrm{ Geo}})               &= \dfrac{1-p}{p}  \\
			\Var(\ZVX_{\mathrm{ Geo}})             &= \dfrac{1-p}{p^2} 
		\end{align*}
	\end{alertblock}
	%
	\begin{alertblock}{Note} 
		In the literature, an alternative model is also used in which the geometrically distributed random variable $\ZVY_{\mathrm{ Geo}}$ gives the number of trials \alert{up to the first success}, i.e., $\ZVY_{\mathrm{ Geo}} \in \MNn$. Trivially, $\ZVY_{\mathrm{ Geo}} = \ZVX_{\mathrm{ Geo}} + 1$.
	\end{alertblock}
	
	\begin{alertblock}{Example}
		For Geometric distrtibuted random variables $\ZVX_{\mathrm{ Geo}}$ with $p = 0.2$ and $p = 0.3$, we obtain the following probability mass functions $f(x)$ [Schira, Statistische Methoden der VWL und BWL, 2021]:
		\begin{center}
			\includegraphics[scale=0.7]{Bilder/GeometricDistrib.png}
		\end{center}
	\end{alertblock}
	
	\begin{alertblock}{Example}
		\begin{itemize}
			\item In Tunis, hailing a taxi is not guaranteed; waiting time is acceptable due to many taxis.
			\item Probability that a taxi arrives within the next minute: $p = 0.2$.
			\item Let $\ZVX_{\mathrm{ Geo}}$ be the number of unsuccessful minutes waited (number of failures before the first success).
		\end{itemize}
	    \begin{center}
	    	\includegraphics[scale=0.7]{Bilder/GeometricDistrib_Ex.png}
	    \end{center}
		
		\vspace{0.2cm}
		
		The probability of having to wait no longer than three minutes is
		\[
		F_{\text{Geo}}(2;0.2) = 0.488.
		\]
		
		The probability of having to wait at least five minutes is
		\[
		P(\ZVX_{\mathrm{ Geo}} > 5) = 1 - F_{\text{Geo}}(5;0.2) = 0.2622.
		\]
		
		The random variable has expected value
		$\E(\ZVX_{\mathrm{ Geo}}) = \frac{q}{p} = \frac{0.8}{0.2} = 4$,
		which means that, on average, one has to wait four full minutes without success.

	\end{alertblock}

\begin{frame}%[allowframebreaks,allowdisplaybreaks]
	\frametitle<presentation>{Exponential Distribution}
	\begin{alertblock}{Model}
		Continuous version of the \emph{Geometric distribution}. The curve of the strictly monotonically decreasing exponential function $e^{-x}$
		encloses, together with the $x$- and $y$-axes in the first quadrant, an area of size $1\,\text{area unit}$ and is therefore suitable 
		as a \emph{density function}. 
	\end{alertblock}
	%
	\begin{alertblock}{Exponential Distribution}
		\begin{eqnarray*}
			f_{\mathrm{ Ex}}(x; \lambda) &=& \lambda \cdot \operatorname{e}^{-\lambda x}; \quad x \geq 0; \; \lambda > 0 \label{eq:11-18} \\
			F_{\mathrm{ Ex}}(x) &=& 1 - \operatorname{e}^{-\lambda x} \\
			\E(\ZVX_{\mathrm{ Ex}}) &=& \dfrac{1}{\lambda} \label{eq:11-18a} \\
			\Var(\ZVX_{\mathrm{ Ex}}) &=& \dfrac{1}{\lambda^2} \label{eq:11-18b}
		\end{eqnarray*}
	\end{alertblock}
	%
	\begin{alertblock}{Note} 
		The exponential distribution is a \emph{memoryless distribution}: let $w$ be an already elapsed waiting time, then:
		$$f_{\mathrm{ Ex}\mid w}(w+x\mid X>w; \lambda) = f_{\mathrm{ Ex}}(x; \lambda)$$
	\end{alertblock}
\end{frame}

\begin{frame}%[allowframebreaks,allowdisplaybreaks]
	\frametitle<presentation>{Rectangular Distribution}
	\begin{alertblock}{Model}
		A continuous random variable $\ZVX_{\mathrm{ R}}$ whose realizations come from an interval $[a,b]$. $\ZVX_{\mathrm{ R}}$ has the property that the probability of obtaining a result in an interval $\Delta x \subseteq [a,b]$ is proportional to the length of this interval, independent of the interval's position within $[a,b]$. This is the continuous version of the \emph{Uniform distribution}.
	\end{alertblock}
	%
	\begin{alertblock}{Rectangular Distribution}
		\begin{eqnarray*}
			f_{\mathrm{ R}}(x) &=& 
			\begin{cases} 
				\dfrac{1}{b-a} & \quad \text{for } a \leq x \leq b \\
				0              & \quad \text{otherwise}
			\end{cases}\label{eq:11-16} \\
			F_{\mathrm{ R}}(x) &=& 
			\begin{cases} 
				0 & \quad \text{for } -\infty \leq x <a \\
				\dfrac{x-a}{b-a} & \quad \text{for } a \leq x \leq b \\
				1              & \quad \text{for } b < x \leq +\infty
			\end{cases} \\
			\E(\ZVX_{\mathrm{ R}}) &=& \dfrac{a+b}{2} \label{eq:11-17a} \\
			\Var(\ZVX_{\mathrm{ R}}) &=& \dfrac{(b-a)^2}{12} \label{eq:11-17b}
		\end{eqnarray*}
	\end{alertblock}
	%
\end{frame}

\begin{frame}[allowframebreaks,allowdisplaybreaks]
	\frametitle<presentation>{Standard Normal Distribution}
	\begin{alertblock}{Model}
		Limit case of the \alert{standardized} \emph{Binomial Distribution} for $n \to \infty$. 
	\end{alertblock}
	%
	\par\smallskip
	
	\begin{alertblock}{Standard Normal Distribution}
		A continuous random variable $\ZVZ$ with density function
		\begin{equation*}
			f_{\mathrm{ St}}(z) = \dfrac{1}{\sqrt{2\cdot \pi}}\operatorname{e}^{-\frac{1}{2}z^2} \label{eq:11-20}
		\end{equation*}
		is called \textbf{standard normally distributed}.
	\end{alertblock}
	%
	\par\smallskip
	
	\begin{alertblock}{Properties}
		\begin{enumerate}
			\item Maximum at $z = 0$
			\item Symmetry with respect to the axis $z \equiv 0$
			\item Inflection points at $z = \pm 1$
			\item We always have $f_{\mathrm{ St}}(z) > 0$ (i.e., $f_{\mathrm{ St}}(z)$ has no zeros)
			\item In the limit: $\lim\limits_{z \to \pm \infty}f_{\mathrm{ St}}(z) = 0$ 
		\end{enumerate}
	\end{alertblock}
	\framebreak
	\begin{alertblock}{}
		\begin{enumerate}
			\setcounter{enumi}{4}
			\item The distribution function
			\begin{equation*}
				F_{\mathrm{ St}}(z) = \frac{1}{\sqrt{2\cdot \pi}}\int\limits_{-\infty}^{z}\operatorname{e}^{-\frac{1}{2}u^2} \,du 
			\end{equation*}
			is \textbf{not available in closed form analytically}. Its values are tabulated and can also be computed via function calls.
			\item Areas:
			%\setcounter{equation}{21}
			\begin{eqnarray*}
				P(\ZVZ \leq z)      &=& \int_{-\infty}^{z} f_{\mathrm{ St}}(u) \,du  = F_{\mathrm{ St}}(z)                 \nonumber \\
				P(a < \ZVZ \leq b)  &=& \int_{a}^{b} f_{\mathrm{ St}}(z) \,dz  = F_{\mathrm{ St}}(b) - F_{\mathrm{ St}}(a) \nonumber \\
				P(-a < \ZVZ \leq a) &=& \int_{-a}^{a} f_{\mathrm{ St}}(z) \,dz  = 2 \cdot F_{\mathrm{ St}}(a) - 1 =: D(a) \label{eq:11-22} \\
			\end{eqnarray*}
			\item Expected value: $\E(\ZVZ) = 0$
			\item Variance: $\Var(\ZVZ) = 1$
		\end{enumerate}
	\end{alertblock}
\end{frame}

\begin{frame}[allowframebreaks,allowdisplaybreaks]
	\frametitle<presentation>{Normal Distribution}
	%
	\begin{alertblock}{Model}
		Obtained from the standard normally distributed random variable $\ZVZ$ by the linear transformation $\ZVX_{\mathrm{ N}} = \mu + \sigma \cdot \ZVZ$ ("de-standardization").
	\end{alertblock}
	%
	\begin{alertblock}{Normal Distribution}
		A continuous random variable $\ZVX_{\mathrm{ N}}$ with density function
		\begin{equation*}
			f_{\mathrm{ N}}(x) = \dfrac{1}{\sigma} f_{\mathrm{ St}}\left(z_x = \dfrac{x-\mu}{\sigma} \right) = \frac{1}{\sigma \cdot \sqrt{2\cdot \pi}}\mye^{-\frac{1}{2}\left( \frac{x-\mu}{\sigma} \right)^2} \label{eq:11-24} 
		\end{equation*}
		is called \textbf{normally distributed}.
	\end{alertblock}
	%%
	\begin{alertblock}{Properties}
		\begin{enumerate}
			\item Expected value and variance:
			\begin{eqnarray*}
				\E(\ZVX_{\mathrm{ N}})  &=& \mu \\
				\Var(\ZVX_{\mathrm{ N}}) &=& \sigma^2 
			\end{eqnarray*}
			\item Maximum at $x = \mu$
			\item Symmetry w.r.t.~$x \equiv \mu$
			\item Inflection points at $x = \mu \pm \sigma$
		\end{enumerate}
	\end{alertblock}
	
	\framebreak
	\begin{alertblock}{}
		\begin{enumerate}
			\setcounter{enumi}{4}
			\item Becomes flatter as $\sigma$ increases:
			\begin{equation*}
				\label{eq:11-25}
				f_{\mathrm{ N}}(x) = \dfrac{1}{\sigma} f_{\mathrm{ St}}\left(z_x = \dfrac{x-\mu}{\sigma} \right)
			\end{equation*}
			\item The distribution function
			\begin{equation*}
				F_{\mathrm{ N}}(x) = \frac{1}{\sigma \cdot \sqrt{2\cdot \pi}}\int\limits_{-\infty}^{x}\mye^{-\frac{1}{2}\left( \frac{u-\mu}{\sigma} \right)^2} \operatorname{d}u = F_{\mathrm{ St}}\left(z_x = \dfrac{x-\mu}{\sigma} \right)
			\end{equation*}
			can be reduced to the standard normal distribution $F_{\mathrm{ St}}(z)$.
			\item Areas:
			%\setcounter{equation}{21}
			\begingroup\small
			\begin{eqnarray*}
				P(\ZVX_{\mathrm{ N}} \leq x)      &=& F_{\mathrm{ St}}\left(z_x = \dfrac{x-\mu}{\sigma} \right)                 \nonumber \\
				P(a < \ZVX_{\mathrm{ N}} \leq b)  &=& F_{\mathrm{ St}}\left(z_b = \dfrac{b-\mu}{\sigma} \right) - F_{\mathrm{ St}}\left(z_a = \dfrac{a-\mu}{\sigma} \right) \nonumber \\
				P(-a < \ZVX_{\mathrm{ N}} \leq a) &=& 2 \cdot F_{\mathrm{ St}}\left(z_a = \dfrac{a-\mu}{\sigma} \right) - 1 \\
			\end{eqnarray*}
			\endgroup
		\end{enumerate}
	\end{alertblock}
